"""Deliberate faults in the M21C1 sound occlusion; which tests fail. Each file is restored.

Usage: py mutate_occlusion.py <repo> <out.json> [name part ...]

Headless tests (tests/test_sound_occlusion.py) run for every fault; the real-window ones
(tests/test_sound_occlusion_gpu.py, MESH_GL_SMOKE=1) for the faults marked GPU as well.
"""
import json
import os
import re
import subprocess
import sys
import time
from pathlib import Path

REPO = Path(sys.argv[1])
OUT = Path(sys.argv[2])
ONLY = sys.argv[3:]
SE, OCC, OPS = "engine/sound_emitters.py", "engine/sound_occlusion.py", "engine/project_ops.py"
HEADLESS = ["tests/test_sound_occlusion.py"]
GPU = ["tests/test_sound_occlusion_gpu.py"]

MUTATIONS = {
    "none": (None, None, None, True),
    "the wall question always answers no (the occlusion calculation removed)": (
        OCC, "    if what is None:\n        return False\n    if what[0] == \"2d\":\n",
        "    if what is not None:\n        return False\n    if what[0] == \"2d\":\n", True),
    "an emitter behind a wall is not turned down": (
        SE, "                targets[live.key] = (level * live.occlusion, side * (1.0 - (1.0 - OCCLUDED_PAN) * behind), d)\n",
        "                targets[live.key] = (level, side * (1.0 - (1.0 - OCCLUDED_PAN) * behind), d)\n", True),
    "its side is not narrowed behind a wall": (
        SE, "                targets[live.key] = (level * live.occlusion, side * (1.0 - (1.0 - OCCLUDED_PAN) * behind), d)\n",

    "it jumps behind a wall and out (no easing)": (

    "every due emitter is asked in one update (no budget)": (
        SE, "        for _asked, _when, _index, live, at in due[:OCCLUSION_ASKS]:\n",
        "        for _asked, _when, _index, live, at in due:\n", False),
    "the question is asked every update (nothing remembered)": (
        SE, "        if asked is not None and asked[0] is here:\n", "        if asked is not None and asked[0] is None:\n", True),
    "emitters out of reach are asked too": (
        SE, "            if d < e.distance:\n                walled.append(", "            if True:\n                walled.append(", False),
    "every emitter minds walls, set to or not (rooms made before this change)": (

        "        if True:\n", False),

results = {}
for name, (file, old, new, gpu) in MUTATIONS.items():
    if ONLY and not any(part in name for part in ONLY):
        continue
    original = None
    try:
        if file is not None:
            path = REPO / file
            original = path.read_bytes()
            text = original.decode("utf-8")
            nl = "\r\n" if "\r\n" in text else "\n"
            flat = text.replace("\r\n", "\n")
            if flat.count(old) != 1:
                results[name] = {"error": f"pattern found {flat.count(old)} times in {file}"}
                print("SKIPPED ", name, results[name]["error"], flush=True)
                original = None
                continue
            path.write_bytes(flat.replace(old, new).replace("\n", nl).encode("utf-8"))
        started = time.perf_counter()
        failed, summaries = [], []
        for tests, env in ((HEADLESS, dict(os.environ)), (GPU, dict(os.environ, MESH_GL_SMOKE="1"))):
            if tests is GPU and not gpu:
                continue
            args = ["-m", "gl"] if tests is GPU else []
            run = subprocess.run([sys.executable, "-m", "pytest", "-o", "addopts=", "-q", "-p", "no:cacheprovider", "-W", "error",
                                  "--strict-markers", *args, *tests], cwd=REPO, env=env, capture_output=True, text=True, timeout=900)
            failed += sorted(set(re.findall(r"^(?:FAILED|ERROR) \S+?::(.+?)(?: - .*)?$", run.stdout, re.M)))
            lines = [line for line in run.stdout.strip().splitlines() if " passed" in line or " failed" in line or " error" in line]
            summaries.append(lines[-1] if lines else run.stderr[-200:])
        results[name] = {"file": file, "caught": bool(failed), "failing": failed, "summary": " | ".join(summaries),
                         "wall_s": round(time.perf_counter() - started, 1)}
    finally:
        if original is not None:
            (REPO / file).write_bytes(original)
    r = results[name]
    print(("CAUGHT " if r["caught"] else "PASSED ") + f" {name}: {r['summary']}", flush=True)
    for test in r["failing"]:
        print("          ", test[:150], flush=True)
    OUT.write_text(json.dumps(results, indent=1), encoding="utf-8")
print(subprocess.run(["git", "status", "--short"], cwd=REPO, capture_output=True, text=True).stdout)
