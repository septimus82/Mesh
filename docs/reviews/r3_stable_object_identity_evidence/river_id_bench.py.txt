"""unique_river_id before (f9febde9) and after (working tree): peak allocation and time per call, same conditions.

    py -3.13 bench_river.py REPO OUT.json

Both versions' source is taken from the files themselves and run against the same engine helpers (``rivers``,
``_retired_here``), in one process, in alternating order. Studio's case: a room with a few rivers, inside an edit.
"""
import ast
import itertools
import json
import subprocess
import sys
import timeit
import tracemalloc
from pathlib import Path

REPO, OUT = Path(sys.argv[1]).resolve(), Path(sys.argv[2])
sys.path.insert(0, str(REPO))
from engine import project_ops as ops  # noqa: E402


def function(source):
    node = next(n for n in ast.parse(source).body if isinstance(n, ast.FunctionDef) and n.name == "unique_river_id")
    space = {"rivers": ops.rivers, "_retired_here": ops._retired_here, "itertools": itertools, "Doc": dict}
    exec(compile(ast.Module([node], []), "unique_river_id", "exec"), space)
    return space["unique_river_id"]


old = function(subprocess.run(["git", "show", "f9febde9700a14fd11d688badb5b556aa00b0cec:engine/project_ops.py"],
                              cwd=REPO, capture_output=True, text=True, check=True, encoding="utf-8").stdout)
new = function((REPO / "engine/project_ops.py").read_text(encoding="utf-8"))
cases = {"no rivers": [], "3 rivers": ["river", "river-2", "river-3"]}
results = {}
with ops.retired_ids({ops.scoped_id("river", "start", "river-4")}), ops.in_room("start"):
    for case, present in cases.items():
        scene = {"settings": {"hd2d": {"rivers": [{"id": r} for r in present]}}}
        assert old(scene) == new(scene)
        row = {"id": new(scene)}
        for name, fn in (("before", old), ("after", new)):
            tracemalloc.start()
            fn(scene)
            row[f"{name} peak bytes"] = tracemalloc.get_traced_memory()[1]
            tracemalloc.stop()
        for name, fn in (("before", old), ("after", new), ("before", old), ("after", new)):
            best = min(timeit.repeat(lambda fn=fn: fn(scene), number=20, repeat=5)) / 20
            row[f"{name} us per call"] = min(row.get(f"{name} us per call", 1e18), round(best * 1e6, 2))
        results[case] = row
text = json.dumps({"python": sys.version.split()[0], "results": results}, indent=1)
print(text)
OUT.write_text(text + "\n", encoding="utf-8")
