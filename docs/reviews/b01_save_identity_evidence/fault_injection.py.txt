def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def git_diff(rel: str) -> str:
    return subprocess.run(["git", "diff", "--", rel], cwd=REPO, capture_output=True, text=True).stdout


def main() -> int:
    bad = 0
    for name, rel, old, new, tests in MUTATIONS:
        path = REPO / rel
        original = path.read_bytes()
        before_hash, before_diff = hashlib.sha256(original).hexdigest(), git_diff(rel)
        text = original.decode("utf-8")
        nl = chr(13) + chr(10) if chr(13) in text else chr(10)
        old, new = old.replace(chr(10), nl), new.replace(chr(10), nl)
        assert text.count(old) == 1, (name, text.count(old))
        path.write_bytes(text.replace(old, new).encode("utf-8"))
        try:
            run = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", "-W", "error", "-x",
                                  "--no-header", *tests], cwd=REPO, capture_output=True, text=True)
        finally:
            path.write_bytes(original)
        failed = [line.split(" - ")[0] for line in run.stdout.splitlines() if line.startswith("FAILED")]
        ok = sha(path) == before_hash and git_diff(rel) == before_diff
        caught = run.returncode != 0 and bool(failed)
        bad += (not caught) + (not ok)
        print(f"{name}: {'CAUGHT' if caught else 'NOT CAUGHT'} by {failed[:1] or run.stdout.splitlines()[-1:]}; "
              f"restored byte-identical: {ok}")
    print("ALL CAUGHT AND RESTORED" if not bad else f"PROBLEMS: {bad}")
    return 1 if bad else 0


if __name__ == "__main__":
    raise SystemExit(main())
