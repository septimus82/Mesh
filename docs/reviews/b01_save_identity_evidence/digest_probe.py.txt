class Health:
    def __init__(self, hp): self.hp = hp
    def saveable_state(self): return {"hp": self.hp, "dead": False}

        self.mesh_name = name; self.mesh_id = mesh_id; self.center_x = x; self.center_y = y
        self.mesh_entity_data = {"id": mesh_id} if mesh_id else {}
        self.mesh_behaviours_runtime = [Health(hp)]; self.mesh_tag = tag
