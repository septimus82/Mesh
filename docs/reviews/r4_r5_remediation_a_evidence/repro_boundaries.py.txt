"""R4/R5 remediation A: does each protected writer hold the lock of the Mesh project its files belong to?

    py -3.13 repro_boundaries.py REPO OUT_JSON

Runs the R4 cases if REPO has R4 (mesh_cli/ai.py `_own_project`), the R5 cases if it has R5 (tools `_own_for_files`).
Disposable scaffolded projects; every Studio / lock holder is its own process (the real OS lock); the writers are
the production code paths (the console's ai_job through ConsoleController, mesh apply-plan / undo-last-plan, the
MCP tools)."""

from __future__ import annotations

import hashlib
import json
import os
import subprocess
import sys
import tempfile
import threading
from pathlib import Path

REPO = Path(sys.argv[1]).resolve()
OUT = Path(sys.argv[2])
sys.path.insert(0, str(REPO))
os.chdir(REPO)

from engine.editor import project_ownership  # noqa: E402
from engine.project_scaffold import create_project  # noqa: E402
from tests.test_studio_project_ownership import StudioProcess  # noqa: E402

TMP = Path(tempfile.mkdtemp(prefix="mesh-r45-"))
HAS_R4 = "_own_project" in (REPO / "mesh_cli/ai.py").read_text(encoding="utf-8")
HAS_R5 = "_own_for_files" in (REPO / "engine/mcp_server/tools.py").read_text(encoding="utf-8")
REGION = "packs/core_regions"
HOLD = ("import sys, time; sys.path.insert(0, sys.argv[1]); from engine.editor import project_ownership as o; "
        "h = o.acquire(sys.argv[2], sys.argv[3]); print('held', flush=True); time.sleep(120)")


def files(root: Path) -> dict[str, str]:
    skip = ("owner.lock", "live_session.json")
    return {p.relative_to(root).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()[:16]
            for p in sorted(root.rglob("*")) if p.is_file() and p.name not in skip}


def diff(a: dict[str, str], b: dict[str, str]) -> dict[str, list[str]]:
    return {"changed": sorted(k for k in a if k in b and a[k] != b[k]), "created": sorted(set(b) - set(a)),
            "deleted": sorted(set(a) - set(b))}


def project(label: str) -> Path:
    root = TMP / label / "game"
    create_project(root, "Game", template_id="blank")
    (root / "assets/data").mkdir(parents=True, exist_ok=True)
    (root / "assets/data/quests.json").write_text('{"quests": {}}\n', encoding="utf-8")
    return root


def studio(root: Path, label: str, *args: str) -> StudioProcess:
    s = StudioProcess(root, *args, log=TMP / f"{label}.log", env={"MESH_TEST_OWNER_WAIT": "0.3"})
    assert s.start["ready"], s.start
    s.ask("edit Unsaved Warden")
    return s


class Holder:
    """Another process owning ``root`` as ``kind`` (the real OS lock)."""

    def __init__(self, root: Path, kind: str) -> None:
        self.proc = subprocess.Popen([sys.executable, "-c", HOLD, str(REPO), str(root), kind], stdout=subprocess.PIPE,
                                     text=True)
        assert self.proc.stdout is not None and self.proc.stdout.readline().strip() == "held"

    def end(self) -> None:
        self.proc.kill()
        self.proc.wait(timeout=30)
        if self.proc.stdout is not None:
            self.proc.stdout.close()


def ai_job(cwd: Path, job: dict) -> list[str]:
    from engine.console_controller import ConsoleController

    path = TMP / f"job_{len(os.listdir(TMP))}.json"
    path.write_text(json.dumps(job), encoding="utf-8")

    class Window:  # a game window (no editor): reload_scene only
        def reload_scene(self) -> None:
            pass

    before = os.getcwd()
    os.chdir(cwd)
    try:
        console = ConsoleController(Window())
        console.execute_command(f"ai_job {path}")
        return console.lines[1:]
    finally:
        os.chdir(before)


def mesh(cwd: Path, *args: str) -> dict:
    run = subprocess.run([sys.executable, "-m", "mesh_cli", *args], cwd=cwd, capture_output=True, text=True,
                         timeout=300, env={**os.environ, "PYTHONPATH": str(REPO)})
    return {"exit": run.returncode, "said": [line for line in run.stdout.splitlines() if "[Mesh]" in line][-4:]}


# ------------------------------------------------------------------------------------------------------- R5
def _after_the_check(work, meanwhile) -> dict:
    """Run ``work`` (an MCP tool call); right after its F1 check (``tools._files_blocked``) answers, pause, run
    ``meanwhile`` (someone takes the project), then let the tool go on."""
    from engine.mcp_server import tools

    checked, go = threading.Event(), threading.Event()
    check = tools._files_blocked

    def check_then_pause(r, t):  # type: ignore[no-untyped-def]
        verdict = check(r, t)
        checked.set()
        go.wait(60)
        return verdict

    tools._files_blocked = check_then_pause
    result: list = []
    thread = threading.Thread(target=lambda: result.append(work()))
    thread.start()
    try:
        assert checked.wait(60)
        extra = meanwhile()
    finally:
        go.set()
        thread.join(60)
        tools._files_blocked = check
    return {"result": {k: result[0].get(k) for k in ("ok", "reason")}, **(extra or {})}


def r5_interior_root(tool: str) -> dict:
    from engine.mcp_server import tools

    root = project(f"r5_interior_{tool}")
    interior = root / REGION
    state: dict = {}

    def work():  # type: ignore[no-untyped-def]
        if tool == "apply_ops":
            return tools.apply_ops([{"type": "set_world_start", "world_path": "worlds/main.json", "start_scene": "attic"}],
                                   root=str(interior), validate=False)
        return tools.create_scene("scenes/attic", root=str(interior))

    def meanwhile():  # type: ignore[no-untyped-def]
        state["studio"] = studio(root, f"r5_interior_{tool}")  # Mesh Studio opens the project, edits, unsaved
        state["before"] = files(root)

    out = _after_the_check(work, meanwhile)
    out.update({"tool": tool, "tool_root": f"<project>/{REGION}", "studio_opened_after_the_check": "the project",
                "files_written_behind_studio": diff(state["before"], files(root)),
                "tool_lock_file_in_the_interior_folder": (interior / ".mesh/owner.lock").exists()})
    state["studio"].close()
    return out


def r5_two_writers_two_dirs() -> dict:
    from engine.mcp_server import tools

    root = project("r5_two")
    interior = root / REGION
    state: dict = {}

    def work():  # type: ignore[no-untyped-def]
        return tools.apply_ops([{"type": "set_world_start", "world_path": "worlds/main.json", "start_scene": "b"}],
                               root=str(interior), validate=False)

    def meanwhile():  # type: ignore[no-untyped-def]
        state["a"] = Holder(root, project_ownership.AUTHORING)  # writer A: owns the project, mid-write
        state["before"] = files(root)

    try:
        out = _after_the_check(work, meanwhile)
        out["owner_while_b_wrote"] = (project_ownership.owned(root, wait=0) or {}).get("kind")
    finally:
        state["a"].end()
    out.update({"writer_a": "another process owning <project> as a writer (mid-write)",
                "writer_b": f"apply_ops with root <project>/{REGION}", "writer_b_wrote": diff(state["before"], files(root)),
                "writer_b_lock_file_in_the_interior_folder": (interior / ".mesh/owner.lock").exists()})
    return out


# ------------------------------------------------------------------------------------------------------- R4
def r4_console_interior() -> dict:
    root = project("r4_console_interior")
    s = studio(root, "r4_console_interior", "nobridge")
    before = files(root)
    said = ai_job(root / REGION, {"operations": [{"type": "add_light", "scene_path": "scenes/start.json",
                                                   "light": {"x": 1, "y": 1, "radius": 9}}]})
    out = {"studio_owns": "the project", "console_cwd": f"<project>/{REGION}", "console_said": said,
           "files_written_behind_studio": diff(before, files(root))}
    s.close()
    return out


def r4_console_into_nested() -> dict:
    root = project("r4_console_nested")
    inner = root / "inner"
    create_project(inner, "Inner", template_id="blank")
    s = studio(inner, "r4_console_nested", "nobridge")
    before = files(inner)
    said = ai_job(root, {"operations": [{"type": "add_light", "scene_path": f"inner/{REGION}/scenes/start.json",
                                          "light": {"x": 1, "y": 1, "radius": 9}}]})
    out = {"studio_owns": "<project>/inner (its own Mesh project)", "console_cwd": "<project>", "console_said": said,
           "nested_files_written_behind_its_studio": diff(before, files(inner))}
    s.close()
    return out


def r4_apply_plan_interior() -> dict:
    root = project("r4_plan_interior")
    s = studio(root, "r4_plan_interior", "nobridge")
    plan = TMP / "plan_interior.json"
    plan.write_text(json.dumps({"wizard": "r45", "version": 1, "inputs": {}, "actions": [
        {"type": "add_npc", "description": "AI adds a Ferryman", "args": {"scene_path": "scenes/start.json",
                                                                          "role": "villager", "name": "Ferryman"}}]}),
                    encoding="utf-8")
    before = files(root)
    ran = mesh(root / REGION, "apply-plan", str(plan))
    out = {"studio_owns": "the project", "cwd": f"<project>/{REGION}", "apply_plan": ran,
           "files_written_behind_studio": diff(before, files(root))}
    s.close()
    return out


def r4_undo_manifest_into_nested() -> dict:
    """A plan applied (Studio closed) modified a room and created a scene in packs/region2; region2 later became its
    own Mesh project, now owned (Studio there). Undo in the outer project."""
    root = project("r4_undo")
    plan = TMP / "plan_undo.json"
    plan.write_text(json.dumps({"wizard": "r45", "version": 1, "inputs": {"pack": "region2"}, "actions": [
        {"type": "create_scene", "description": "a scene in region2",
         "args": {"path": "packs/region2/scenes/cellar.json", "template": "empty"}}]}), encoding="utf-8")
    applied = mesh(root, "apply-plan", "--no-lint", str(plan))
    # the same backup also restores a room of this project (as a plan touching both packs would record it)
    backups = sorted((root / ".mesh/plan_backups").iterdir())
    manifest_path = backups[-1] / "manifest.json"
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    room = f"{REGION}/scenes/start.json"
    saved = backups[-1] / "files" / room
    saved.parent.mkdir(parents=True, exist_ok=True)
    saved.write_bytes((root / room).read_bytes())
    manifest["modified"].append(room)
    manifest_path.write_text(json.dumps(manifest), encoding="utf-8")
    (root / room).write_text((root / room).read_text(encoding="utf-8").replace("{", "{\"edited_after\": 1, ", 1), encoding="utf-8")
    nested = root / "packs/region2"
    (nested / "config.json").write_text(json.dumps({"title": "Region 2", "world_file": "worlds/main.json"}), encoding="utf-8")
    holder = Holder(nested, project_ownership.STUDIO)  # Mesh Studio has region2 open
    before_outer, before_nested = files(root / REGION), files(nested)
    try:
        undone = mesh(root, "undo-last-plan")
    finally:
        holder.end()
    return {"plan_applied": applied["exit"], "manifest": {"created": manifest["created"], "modified": manifest["modified"]},
            "nested_now_owned_by": "Mesh Studio (a separate process)", "undo": undone,
            "nested_files_changed": diff(before_nested, files(nested)),
            "outer_files_changed": diff(before_outer, files(root / REGION))}


def r4_undo_failure_reporting() -> dict:
    root = project("r4_undo_fail")
    backup = root / ".mesh/plan_backups/1000_corrupt"
    backup.mkdir(parents=True)  # a backup with no manifest: the restore fails
    return {"undo_with_corrupt_backup": mesh(root, "undo-last-plan")}


def main() -> None:
    head = subprocess.run(["git", "rev-parse", "HEAD"], cwd=REPO, capture_output=True, text=True).stdout.strip()
    dirty = bool(subprocess.run(["git", "status", "--porcelain", "--", "engine", "mesh_cli"], cwd=REPO,
                                capture_output=True, text=True).stdout.strip())
    out: dict = {"head": head, "uncommitted_changes": dirty, "has_r4": HAS_R4, "has_r5": HAS_R5}
    if HAS_R5:
        out["r5_mcp_root_is_an_interior_folder"] = {t: r5_interior_root(t) for t in ("apply_ops", "create_scene")}
        out["r5_two_writers_from_two_folders"] = r5_two_writers_two_dirs()
    if HAS_R4:
        out["r4_console_from_an_interior_folder"] = r4_console_interior()
        out["r4_console_into_a_nested_project"] = r4_console_into_nested()
        out["r4_apply_plan_from_an_interior_folder"] = r4_apply_plan_interior()
        out["r4_undo_manifest_into_a_nested_project"] = r4_undo_manifest_into_nested()
        out["r4_undo_failure_reporting"] = r4_undo_failure_reporting()
    text = json.dumps(out, indent=2, default=str)
    OUT.write_text(text + "\n", encoding="utf-8")
    print(text)


if __name__ == "__main__":
    main()
